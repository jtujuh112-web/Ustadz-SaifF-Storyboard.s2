const FORBIDDEN_CTA_PHRASES = [
  'link di bio',
  'keranjang kuning',
  'lagi po',
  'lagi diskon',
  'kirim sekarang',
  'lagi promo'
];

function sanitizeForbiddenCTA(text) {
  if (!text) return text;
  let cleaned = text;
  const replacements = [
    { regex: /cek\s+link\s+di\s+bio|link\s+di\s+bio/gi, replacement: 'coba cek detail produknya' },
    { regex: /cek\s+keranjang\s+kuning|di\s+keranjang\s+kuning|keranjang\s+kuning/gi, replacement: 'pilihan produk ini' },
    { regex: /lagi\s+po\b|sedang\s+po\b/gi, replacement: 'tersedia untuk kebutuhanmu' },
    { regex: /lagi\s+diskon|sedang\s+diskon/gi, replacement: 'layak banget dipertimbangkan' },
    { regex: /kirim\s+sekarang/gi, replacement: 'bisa masuk daftar pilihanmu' },
    { regex: /lagi\s+promo|sedang\s+promo/gi, replacement: 'punya fitur yang fungsional' }
  ];
  replacements.forEach(({ regex, replacement }) => {
    cleaned = cleaned.replace(regex, replacement);
  });
  return cleaned;
}

function checkContainsForbiddenCTA(text) {
  if (!text) return [];
  const lower = text.toLowerCase();
  return FORBIDDEN_CTA_PHRASES.filter((phrase) => lower.includes(phrase));
}

function estimateSpeechMetrics(dialogue, language) {
  if (!dialogue || language === 'none') {
    return {
      wordCount: 0,
      estimatedSeconds: 0,
      status: 'muted',
      label: 'Tanpa Dialog (Visual & Ambient Audio 10.0 dtk)'
    };
  }
  const words = dialogue
    .trim()
    .split(/\s+/)
    .filter((w) => w.length > 0);
  const wordCount = words.length;
  // Natural energetic TikTok affiliate pace in ID/MY is ~2.4 words/sec + ~1.2s total pauses/action breathing room
  const speakingTime = wordCount / 2.4;
  const pauseBuffer = 1.2;
  const totalEstimated = Number((speakingTime + pauseBuffer).toFixed(1));

  let status = 'optimal';
  let label = 'Pas untuk 10 Detik (Ideal 20–25 kata)';
  if (wordCount < 16) {
    status = 'short';
    label = 'Agak Pendek (<16 kata) — Banyak ruang jeda visual';
  } else if (wordCount > 27) {
    status = 'long';
    label = 'Terlalu Panjang (>27 kata) — Risiko terpotong di 10 detik';
  }

  return {
    wordCount,
    estimatedSeconds: totalEstimated,
    status,
    label
  };
}

// Convert Base64 PCM16 from Gemini TTS to playable WAV Blob
function pcmBase64ToWavUrl(base64Audio, sampleRate = 24000) {
  const binaryString = atob(base64Audio);
  const len = binaryString.length;
  const bytes = new Uint8Array(len);
  for (let i = 0; i < len; i++) {
    bytes[i] = binaryString.charCodeAt(i);
  }
  const pcm16 = new Int16Array(bytes.buffer);
  const numChannels = 1;
  const byteRate = sampleRate * numChannels * 2;
  const blockAlign = numChannels * 2;
  const wavBuffer = new ArrayBuffer(44 + pcm16.byteLength);
  const view = new DataView(wavBuffer);

  const writeString = (offset, str) => {
    for (let i = 0; i < str.length; i++) {
      view.setUint8(offset + i, str.charCodeAt(i));
    }
  };

  writeString(0, 'RIFF');
  view.setUint32(4, 36 + pcm16.byteLength, true);
  writeString(8, 'WAVE');
  writeString(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, numChannels, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, byteRate, true);
  view.setUint16(32, blockAlign, true);
  view.setUint16(34, 16, true);
  writeString(36, 'data');
  view.setUint32(40, pcm16.byteLength, true);

  const pcmBytes = new Uint8Array(wavBuffer, 44);
  pcmBytes.set(bytes);

  const blob = new Blob([wavBuffer], { type: 'audio/wav' });
  return URL.createObjectURL(blob);
}

async function fetchGeminiWithRetry(url, payload, maxRetries = 3) {
  let delay = 1000;
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (response.ok) {
        return await response.json();
      }

      const errData = await response.json().catch(() => ({}));
      if (response.status >= 400 && response.status < 500 && response.status !== 429) {
        throw new Error(errData?.error?.message || `HTTP Error ${response.status}`);
      }
    } catch (err) {
      if (attempt === maxRetries - 1) throw err;
    }
    await new Promise((r) => setTimeout(r, delay));
    delay *= 2;
  }
  throw new Error('Gagal menghubungi server AI setelah beberapa percobaan.');
}
